1. 从 Windows 连接

先在 PowerShell 进入服务器，再以只读会话连接数据库。

ssh elma-gohan

# 以下命令在服务器中执行
source /etc/elma-gohan/elma-gohan.env

PGOPTIONS='-c default_transaction_read_only=on' \
PGPASSWORD="$DB_PASSWORD" \
psql -X \
  -h "$DB_HOST" \
  -p "$DB_PORT" \
  -U "$DB_USERNAME" \
  -d "$DB_NAME"

# 退出 psql 后清理当前 shell 中的密码变量
unset DB_PASSWORD

2. 进入 psql 后

反斜杠开头的是 psql 命令，不加分号；SQL 要加分号。

\conninfo
SHOW default_transaction_read_only;
SHOW timezone;

\dt                 -- 查看所有表
\dv                 -- 查看所有视图
\d+ recommendation_log
\x auto             -- 宽记录自动纵向显示
\pset pager off     -- 关闭分页器

SELECT count(*) FROM recommendation_log;

\q                  -- 退出
生产库注意：确认 default_transaction_read_only 为 on。不要执行 UPDATE、DELETE、TRUNCATE、DROP；查看 JSON 或明细时先加 LIMIT。

匿名用户规模与时间范围

SELECT
  count(*) AS recommendation_count,
  count(DISTINCT anonymous_user_id) AS anonymous_ids,
  min(created_at) AS first_seen,
  max(created_at) AS last_seen
FROM recommendation_log;

每天新增、活跃和请求量

WITH first_seen AS (
  SELECT anonymous_user_id, min(created_at)::date AS first_date
  FROM recommendation_log
  GROUP BY anonymous_user_id
), daily AS (
  SELECT created_at::date AS day,
         count(*) AS requests,
         count(DISTINCT anonymous_user_id) AS active_ids
  FROM recommendation_log
  GROUP BY created_at::date
)
SELECT d.day, d.requests, d.active_ids,
       count(f.anonymous_user_id) AS new_ids
FROM daily d
LEFT JOIN first_seen f ON f.first_date = d.day
GROUP BY d.day, d.requests, d.active_ids
ORDER BY d.day;

粗粒度地理网格（约 10 公里，避免精确定位）

SELECT
  round((request_condition_json ->> 'latitude')::numeric, 1) AS lat_grid,
  round((request_condition_json ->> 'longitude')::numeric, 1) AS lon_grid,
  count(*) AS requests,
  count(DISTINCT anonymous_user_id) AS anonymous_ids
FROM recommendation_log
GROUP BY lat_grid, lon_grid
ORDER BY anonymous_ids DESC, requests DESC;

查看最近推荐结果，不显示精确坐标和完整 UUID

SELECT
  rl.created_at,
  left(rl.anonymous_user_id::text, 8) || '…' AS user_tag,
  r.name AS current_restaurant,
  rl.candidate_count,
  rl.risk_score,
  rl.low_regret_score,
  rl.selection_mode
FROM recommendation_log rl
JOIN restaurant r ON r.id = rl.current_restaurant_id
ORDER BY rl.created_at DESC
LIMIT 20;
